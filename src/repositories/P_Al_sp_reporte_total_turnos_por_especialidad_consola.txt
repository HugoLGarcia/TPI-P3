DELIMITER $$
CREATE DEFINER=`root`@`localhost` PROCEDURE `sp_reporte_total_turnos_por_especialidad_consola`()
BEGIN
	SELECT
    	e.id_especialidad,
        e.nombre AS especialidad,      
    	COUNT(tr.id_turno_reserva) AS cantidad_turnos,
        COALESCE(SUM(tr.valor_total), 0) AS facturacion_total
	FROM especialidades AS e
	LEFT JOIN medicos AS m
    	ON e.id_especialidad = m.id_especialidad
	LEFT JOIN turnos_reservas AS tr
    	ON m.id_medico = tr.id_medico
        AND tr.activo = 1
    WHERE
    	e.activo = 1
    GROUP BY
    	e.nombre
    ORDER BY
    	cantidad_turnos DESC;
END$$
DELIMITER ;
